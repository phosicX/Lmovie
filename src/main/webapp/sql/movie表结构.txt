CREATE TABLE movies (
	id varchar(20) PRIMARY KEY,
	title_zh varchar(100) NOT NULL,
	title_intl varchar(200),
	YEAR int,
	poster_url varchar(200),
	director varchar(50),
	writers text,
	actors text,
	genres varchar(50),
	country varchar(50),
	LANGUAGE varchar(50),
	release_date varchar(100),
	duration varchar(50),
	douban_score varchar(10),
	douban_url varchar(200),
	imdb_score varchar(10),
	imdb_url varchar(200),
	intro text,
	TYPE enum('movie', 'tv_drama', 'anime') DEFAULT 'movie',
	is_active boolean DEFAULT TRUE,
	created_at timestamp DEFAULT CURRENT_TIMESTAMP(),
	updated_at timestamp DEFAULT CURRENT_TIMESTAMP() ON UPDATE CURRENT_TIMESTAMP()
);
